Web Automation: Element Interaction (TextBox, Radio, Checkbox,
Dropdown, Alert, Frames) & Synchronization
Develop a script to automate interaction with various web elements (textbox, radio button,
checkbox, dropdown, alert, and frames). Implement implicit and explicit waits for synchronization
handling.

from playwright.sync_api import sync_playwright, expect
def test_web_elements_and_synchronization():
with sync_playwright() as p:
browser = p.chromium.launch(channel="chrome", headless=False)
context = browser.new_context()
page = context.new_page()
try:
print("Navigating to test automation practice page...")
page.goto("https://rahulshettyacademy.com/AutomationPractice/")
# 1. SYNCHRONIZATION HANDLING
page.set_default_timeout(5000)
# 2. TEXTBOX INTERACTION
print("Interacting with Textbox...")
name_input = page.locator("#name")
name_input.fill("Soham")
expect(name_input).to_have_value("Soham")
# 3. RADIO BUTTON INTERACTION
print("Interacting with Radio Button...")
radio_btn = page.locator("input[value='radio2']")
radio_btn.click()
expect(radio_btn).to_be_checked()
