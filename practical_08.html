Web Automation: Dropdown Enumeration, Default Value Validation &
Selection Testing
Aim
Automate the process of identifying a list or dropdown box on a web page and determine the total
number of items. Validate default selected value and perform selection testing.
Steps:-
1. Open Notepad and type this code and save file as Practical8.py

from playwright.sync_api import sync_playwright, expect
def test_dropdown_and_combo_verification():
with sync_playwright() as p:
browser = p.chromium.launch(channel="chrome", headless=False)
context = browser.new_context()
page = context.new_page()
try:
print("Navigating to test automation practice page...")
page.goto("https://rahulshettyacademy.com/AutomationPractice/")
dropdown_locator = page.locator("#dropdown-class-example")
# 1. TOTAL NUMBER OF ITEMS DETERMINATION
print("Counting total items in the dropdown...")
options_locator = dropdown_locator.locator("option")
total_items = options_locator.count()
print(f"? Total items found in dropdown: {total_items}")
for i in range(total_items):
print(f" - Item {i}: {options_locator.nth(i).inner_text()} (Value:
{options_locator.nth(i).get_attribute('value')})")
# 2. DEFAULT SELECTED VALUE VALIDATION
print("\nValidating default selected value...")
default_value = dropdown_locator.input_value()
print(f"Default selected value attribute: '{default_value}'")
expect(dropdown_locator).to_have_value("")
